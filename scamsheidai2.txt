function analyzeContent(raw,mode){

const t=raw.toLowerCase();

let points=0;
let reasons=[];
let signals=[];

checks.forEach(c=>{

if(c.words.some(w=>t.includes(w))){

points+=c.points;

reasons.push(c.msg);

const found =
c.words.find(w=>t.includes(w));

if(found)
signals.push(found);

}

});


if(mode==="link"){

analyzeUrl(
raw,
reasons,
signals,
value=>points+=value
);

}


if(mode==="email"){

analyzeEmail(
raw,
reasons,
signals,
value=>points+=value
);

}


if(mode==="phone"){

analyzePhone(
raw,
reasons,
signals,
value=>points+=value
);

}


if(
/bit\.ly|tinyurl\.com|t\.co\/|shorturl|cutt\.ly|is\.gd/i
.test(raw)
){

points+=12;

reasons.push(
"uses a shortened link that hides the destination"
);

signals.push("shortened URL");

}


if(
/free|claim now|limited offer|selected customer|lucky customer/i
.test(t)
){

points+=10;

reasons.push(
"uses a promotional or reward-style message"
);

signals.push("reward language");

}


if(
/https?:\/\/|www\./i.test(raw) &&
/otp|password|pin|cvv|payment|pay now|upi/i.test(t)
){

points+=15;

reasons.push(
"combines a link with a request for private information or money"
);

signals.push("link + sensitive request");

}


points=Math.min(100,points);

const level =
points>=60
?"HIGH RISK"
:points>=30
?"MEDIUM RISK"
:"LOW RISK";

const confidence =
Math.min(
98,
Math.max(
68,
72+Math.round(points*0.22)
)
);


const actionText =

points>=60

? "Do not click links, send money, or share OTPs/passwords. Contact the organization using an official website or phone number."

: points>=30

? "Pause. Verify the sender independently and do not share sensitive information until you confirm the request."

: "Remain cautious. Avoid unexpected links and verify important requests through an official source.";


return {

points,
level,
confidence,
reasons:[...new Set(reasons)],
signals:[...new Set(signals)],
actionText

};

}


function analyzeUrl(
raw,
reasons,
signals,
addPoints
){

const urls =
raw.match(
/(?:https?:\/\/|www\.)[^\s<>"']+/gi
)
||
raw.match(
/\b[a-z0-9.-]+\.(?:com|net|org|in|co|io|app|xyz|info|biz)(?:\/[^\s<>"']*)?/gi
)
||[];


if(urls.length===0){

addPoints(10);

reasons.push(
"the input does not look like a complete website address"
);

signals.push("invalid/incomplete URL");

return;

}


const url =
urls[0].replace(/[),.!?]+$/,"");

signals.push("URL detected");

let normalized=url;

if(!/^https?:\/\//i.test(normalized))
normalized="https://"+normalized;


try{

const u=new URL(normalized);

const host=u.hostname.toLowerCase();


if(
/^\d{1,3}(?:\.\d{1,3}){3}$/.test(host)
){

addPoints(25);

reasons.push(
"uses an IP address instead of a normal website name"
);

signals.push("IP address");

}


if(host.includes("xn--")){

addPoints(25);

reasons.push(
"uses a special encoded domain that may imitate another website"
);

signals.push("look-alike domain");

}


if(normalized.includes("@")){

addPoints(25);

reasons.push(
"contains an unusual @ symbol that can hide the real destination"
);

signals.push("@ in URL");

}


if(normalized.length>100){

addPoints(8);

reasons.push(
"the web address is unusually long"
);

signals.push("long URL");

}


if((host.match(/\./g)||[]).length>=4){

addPoints(8);

reasons.push(
"has many subdomains, which can make a website address harder to recognize"
);

signals.push("many subdomains");

}


if(
/login|verify|secure|account|wallet|bonus|reward|prize|claim|update|kyc|banking/i
.test(normalized)
){

addPoints(12);

reasons.push(
"contains words commonly used in fake login, verification or reward links"
);

signals.push("suspicious URL words");

}


if(
/bit\.ly|tinyurl\.com|t\.co\/|shorturl|cutt\.ly|is\.gd/i
.test(host)
){

addPoints(15);

reasons.push(
"uses a URL-shortening service that hides the final destination"
);

signals.push("shortened URL");

}


if(u.protocol==="http:"){

addPoints(10);

reasons.push(
"does not use HTTPS"
);

signals.push("HTTP");

}


if(
/\.(xyz|top|click|buzz|club|work|tk|ml|ga|cf|gq)$/i
.test(host)
){

addPoints(10);

reasons.push(
"uses a domain ending that deserves extra caution"
);

signals.push("unusual domain ending");

}


const brands=[
"google",
"microsoft",
"apple",
"amazon",
"paypal",
"sbi",
"hdfc",
"icici",
"axis",
"phonepe",
"paytm",
"flipkart",
"instagram",
"facebook",
"whatsapp"
];

const brandFound =
brands.find(b=>host.includes(b));

if(
brandFound &&
!isLikelyOfficialDomain(host,brandFound)
){

addPoints(25);

reasons.push(
appears to imitate ${brandFound} using a different website address
);

signals.push("possible impersonation");

}

}
catch(e){

addPoints(15);

reasons.push(
"the website address could not be parsed normally"
);

signals.push("invalid URL");

}

}


function isLikelyOfficialDomain(
host,
brand
){

const official={

google:["google.com","google.co.in"],
microsoft:["microsoft.com"],
apple:["apple.com"],
amazon:["amazon.com","amazon.in"],
paypal:["paypal.com"],
sbi:["sbi.co.in"],
hdfc:["hdfcbank.com"],
icici:["icicibank.com"],
axis:["axisbank.com"],
phonepe:["phonepe.com"],
paytm:["paytm.com"],
flipkart:["flipkart.com"],
instagram:["instagram.com"],
facebook:["facebook.com","fb.com"],
whatsapp:["whatsapp.com"]

};

return (official[brand]||[])
.some(
d=>host===d || host.endsWith("."+d)
);

}


function analyzeEmail(
raw,
reasons,
signals,
addPoints
){

const fromMatch =
raw.match(/(?:from|sender)\s*:\s*([^\n]+)/i);

const subjectMatch =
raw.match(/subject\s*:\s*([^\n]+)/i);

const replyMatch =
raw.match(/reply[- ]?to\s*:\s*([^\n]+)/i);

const from =
fromMatch
?fromMatch[1].trim()
:"";

const subject =
subjectMatch
?subjectMatch[1].trim()
:"";

const replyTo =
replyMatch
?replyMatch[1].trim()
:"";


if(!from && !subject){

addPoints(5);

signals.push("email body only");

}


const emailDomains =
[...raw.matchAll(
/[A-Z0-9._%+-]+@([A-Z0-9.-]+\.[A-Z]{2,})/gi
)]
.map(m=>m[1].toLowerCase());

const uniqueDomains =
[...new Set(emailDomains)];


if(uniqueDomains.length>1){

addPoints(15);

reasons.push(
"the email contains different sender or reply domains"
);

signals.push("domain mismatch");

}


if(replyTo && from){

const fromDomain =
(from.match(
/@([a-z0-9.-]+\.[a-z]{2,})/i
)||[])[1]?.toLowerCase();

const replyDomain =
(replyTo.match(
/@([a-z0-9.-]+\.[a-z]{2,})/i
)||[])[1]?.toLowerCase();

if(
fromDomain &&
replyDomain &&
fromDomain!==replyDomain
){

addPoints(25);

reasons.push(
"the Reply-To address uses a different domain from the sender"
);

signals.push("Reply-To mismatch");

}

}


if(
from &&
/@(gmail|outlook|hotmail|yahoo|protonmail)\./i.test(from) &&
/bank|support|security|official|admin|customer|account/i.test(
subject+" "+raw
)
){

addPoints(18);

reasons.push(
"a personal email address appears to be pretending to represent an organization"
);

signals.push(
"personal email impersonation"
);

}


if(
subject &&
/urgent|immediately|final notice|account suspended|verify now|action required|payment due/i.test(subject)
){

addPoints(18);

reasons.push(
"the subject line uses pressure or fear"
);

signals.push("urgent subject");

}


if(
/\b(attachment|attached|invoice|receipt|document|pdf|zip|apk|exe)\b/i.test(raw)
){

addPoints(10);

reasons.push(
"asks you to open or handle an attachment"
);

signals.push("attachment");

}


if(
/https?:\/\/|www\./i.test(raw)
){

addPoints(8);

reasons.push(
"contains a web link"
);

signals.push("email link");

analyzeUrl(
raw,
reasons,
signals,
addPoints
);

}


if(
/dear customer|dear user|valued customer|security team|verify your account|confirm your identity/i.test(raw)
){

addPoints(10);

reasons.push(
"uses generic account or security language often seen in phishing emails"
);

signals.push(
"generic phishing language"
);

}

}


function analyzePhone(
raw,
reasons,
signals,
addPoints
){

const phone =
raw.match(
/(?:\+91[\s-]?)?[6-9]\d{9}/
);


if(phone){

signals.push(
"phone number detected"
);

}
else{

addPoints(5);

signals.push(
"phone number not clearly detected"
);

}


if(
/otp|password|cvv|pin|bank|kyc|account blocked|police|lottery|prize|payment|upi/i.test(raw)
){

addPoints(18);

reasons.push(
"the caller or message mentions sensitive information, money, banking, or authority pressure"
);

signals.push(
"high-risk call topic"
);

}


if(
/share|tell me|send me|install|screen share|remote|anydesk|teamviewer/i.test(raw)
){

addPoints(25);

reasons.push(
"asks you to share information, install software, or give remote access"
);

signals.push(
"remote-access request"
);

}

}

function reasonToSimple(reason){

const map={

"requests an OTP or verification code":{
title:"It asks for a secret code",
text:"Never give an OTP or verification code to someone through a message or phone call."
},

"requests sensitive credentials":{
title:"It asks for private information",

},

"uses an unexpected prize or reward claim":{
title:"It promises a prize or money",
text:"Unexpected prizes are a common trick. Don't pay a fee or share private information to receive a prize."
},

"creates urgency or fear":{
title:"It is trying to rush or scare you",
text:"Scammers often say 'act now' or 'your account will be blocked' so you don't have time to think."
},

"asks you to follow or verify through a link":{
title:"It wants you to use a link",
text:"Don't open an unexpected link. Open the official app or website yourself instead."
},

"mentions money or a payment request":{
title:"It asks for money",
text:"Stop and verify who is asking and why. Never pay because someone is pressuring you."
},

"uses account, KYC or authority-based pressure":{
title:"It uses fear or authority",
text:"Threats about KYC, police, account blocking, or legal action can be used to pressure you."
},

"mentions high-risk payment methods":{
title:"It suggests an unusual payment method",
text:"Be extra careful with requests for gift cards or cryptocurrency."
},

"contains a clickable web address":{
title:"There is a web link",
text:"A link can lead to a fake website. If you need to check something, type the official website yourself."
},

"uses a shortened or unusual URL pattern":{
title:"The link looks unusual",
text:"Shortened or unfamiliar links can hide where they really go. Avoid opening them."
}

};

return map[reason]
||{
title:"Warning sign detected",
text:reason
};

}

}
